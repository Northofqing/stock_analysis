### 2.1 Factor Trait

```rust
// src/backtest/factor/trait.rs

use chrono::NaiveDate;
use serde::{Deserialize, Serialize};

/// 因子方向
#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
pub enum FactorDirection {
    Ascending,  // 值越大越好 (如 ROE)
    Descending, // 值越小越好 (如 PE)
}

/// 因子计算结果
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FactorValue {
    pub code: String,
    pub date: NaiveDate,
    pub value: f64,
    pub valid: bool,           // 是否有效 (避免 NaN/Inf)
}

/// 因子 trait
pub trait Factor: Send + Sync + std::fmt::Debug {
    /// 因子名称
    fn name(&self) -> &str;

    /// 因子方向
    fn direction(&self) -> FactorDirection;

    /// 计算单只股票的因子值
    fn compute(&self, code: &str, date: NaiveDate, kline: &KlineData) -> FactorValue;

    /// 批量计算 (默认逐个，可优化)
    fn compute_batch(
        &self,
        data: &[(String, NaiveDate, KlineData)],
    ) -> Vec<FactorValue> {
        data.iter()
            .map(|(code, date, kline)| self.compute(code, *date, kline))
            .collect()
    }
}

/// 组合因子 (多个因子加权)
#[derive(Debug, Clone)]
pub struct CompositeFactor {
    pub name: String,
    pub factors: Vec<(Box<dyn Factor>, f64)>, // (因子, 权重)
    pub direction: FactorDirection,
}

impl CompositeFactor {
    pub fn new(name: &str, factors: Vec<(Box<dyn Factor>, f64)>) -> Self {
        let direction = factors
            .first()
            .map(|(f, _)| f.direction())
            .unwrap_or(FactorDirection::Ascending);
        Self {
            name: name.to_string(),
            factors,
            direction,
        }
    }
}

impl Factor for CompositeFactor {
    fn name(&self) -> &str {
        &self.name
    }

    fn direction(&self) -> FactorDirection {
        self.direction
    }

    fn compute(&self, code: &str, date: NaiveDate, kline: &KlineData) -> FactorValue {
        let mut weighted_sum = 0.0;
        let mut total_weight = 0.0;

        for (factor, weight) in &self.factors {
            let fv = factor.compute(code, date, kline);
            if fv.valid {
                let sign = match factor.direction() {
                    FactorDirection::Ascending => 1.0,
                    FactorDirection::Descending => -1.0,
                };
                weighted_sum += fv.value * sign * weight;
                total_weight += weight;
            }
        }

        FactorValue {
            code: code.to_string(),
            date,
            value: if total_weight > 0.0 {
                weighted_sum / total_weight
            } else {
                0.0
            },
            valid: total_weight > 0.0,
        }
    }
}
```

### 2.3 因子统计 (IC/IR)

```rust
// src/backtest/factor/stats.rs

use super::FactorValue;
use chrono::NaiveDate;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;

/// 因子有效性统计
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FactorStats {
    /// 因子名
    pub name: String,
    /// 日期
    pub date: NaiveDate,
    /// IC (信息系数)
    pub ic: f64,
    /// IC 序列 (用于计算 IR)
    pub ic_series: Vec<f64>,
    /// IR (IC均值 / IC标准差)
    pub ir: f64,
    /// 衰减率 (最近60天 vs 历史)
    pub decay_rate: f64,
    /// 样本数
    pub count: usize,
}

impl FactorStats {
    /// 计算 Rank IC (更鲁棒)
    pub fn rank_ic(values: &[f64], returns: &[f64]) -> f64 {
        if values.len() != returns.len() || values.len() < 10 {
            return 0.0;
        }

        // 计算 rank
        let n = values.len();
        let mut value_ranks: Vec<(usize, usize)> = values
            .iter()
            .enumerate()
            .map(|(i, v)| (i, Self::rank_position(values, *v)))
            .collect();
        let mut return_ranks: Vec<(usize, usize)> = returns
            .iter()
            .enumerate()
            .map(|(i, r)| (i, Self::rank_position(returns, *r)))
            .collect();

        // 按原始索引排序
        value_ranks.sort_by_key(|(i, _)| *i);
        return_ranks.sort_by_key(|(i, _)| *i);

        // 计算 rank IC = Pearson(值rank, 收益rank)
        let vr: Vec<f64> = value_ranks.iter().map(|(_, r)| *r as f64).collect();
        let rr: Vec<f64> = return_ranks.iter().map(|(_, r)| *r as f64).collect();

        Self::pearson(&vr, &rr)
    }

    /// Pearson 相关系数
    pub fn pearson(x: &[f64], y: &[f64]) -> f64 {
        if x.len() != y.len() || x.len() < 2 {
            return 0.0;
        }

        let n = x.len() as f64;
        let mean_x = x.iter().sum::<f64>() / n;
        let mean_y = y.iter().sum::<f64>() / n;

        let mut num = 0.0;
        let mut den_x = 0.0;
        let mut den_y = 0.0;

        for i in 0..x.len() {
            let dx = x[i] - mean_x;
            let dy = y[i] - mean_y;
            num += dx * dy;
            den_x += dx * dx;
            den_y += dy * dy;
        }

        if den_x == 0.0 || den_y == 0.0 {
            return 0.0;
        }

        num / (den_x.sqrt() * den_y.sqrt())
    }

    fn rank_position(arr: &[f64], val: f64) -> usize {
        arr.iter().filter(|&&x| x < val).count() + 1
    }

    /// 更新 IC 序列
    pub fn update_ic(&mut self, new_ic: f64) {
        self.ic = new_ic;
        self.ic_series.push(new_ic);

        // 保持最近 60 天
        if self.ic_series.len() > 60 {
            self.ic_series.remove(0);
        }

        // 计算 IR
        if self.ic_series.len() >= 20 {
            let mean = self.ic_series.iter().sum::<f64>() / self.ic_series.len() as f64;
            let std = (self.ic_series.iter()
                .map(|x| (x - mean).powi(2))
                .sum::<f64>() / self.ic_series.len() as f64)
            .sqrt();
            self.ir = if std > 0.0 { mean / std } else { 0.0 };
        }

        // 计算衰减率
        if self.ic_series.len() >= 60 {
            let recent: f64 = self.ic_series.iter().rev().take(20).sum::<f64>() / 20.0;
            let older: f64 = self.ic_series.iter().take(40).sum::<f64>() / 40.0;
            self.decay_rate = if older.abs() > 0.001 {
                (recent - older).abs() / older.abs()
            } else {
                0.0
            };
        }
    }
}

/// IC/IR 批量计算器
pub struct ICRollingCalculator {
    stats: HashMap<String, FactorStats>,
    window: usize, // 滚动窗口
}

impl ICRollingCalculator {
    pub fn new(window: usize) -> Self {
        Self {
            stats: HashMap::new(),
            window,
        }
    }

    /// 计算某因子在某日的 IC
    pub fn calculate_ic(
        &mut self,
        factor_name: &str,
        date: NaiveDate,
        values: &[FactorValue],
        returns: &[f64], // 对应收益率
    ) -> f64 {
        if values.len() != returns.len() || values.len() < 10 {
            return 0.0;
        }

        let valid_values: Vec<f64> = values.iter().map(|v| v.value).collect();
        let ic = FactorStats::rank_ic(&valid_values, returns);

        // 更新统计
        let stats = self.stats.entry(factor_name.to_string()).or_insert_with(|| {
            FactorStats {
                name: factor_name.to_string(),
                date,
                ic: 0.0,
                ic_series: Vec::new(),
                ir: 0.0,
                decay_rate: 0.0,
                count: 0,
            }
        });
        stats.date = date;
        stats.count += values.len();
        stats.update_ic(ic);

        ic
    }

    /// 获取因子统计
    pub fn get_stats(&self, factor_name: &str) -> Option<&FactorStats> {
        self.stats.get(factor_name)
    }

    /// 判断因子是否失效
    pub fn is_factor_decay(&self, factor_name: &str) -> bool {
        self.stats
            .get(factor_name)
            .map(|s| s.decay_rate > 0.3) // 衰减30%认为失效
            .unwrap_or(false)
    }
}
```

### 2.4 因子管理器

```rust
// src/backtest/factor/library.rs

use super::{Factor, FactorValue, ICRollingCalculator};
use std::collections::HashMap;

/// 因子库管理器
pub struct FactorLibrary {
    factors: HashMap<String, Box<dyn Factor>>,
    ic_calculator: ICRollingCalculator,
}

impl FactorLibrary {
    pub fn new() -> Self {
        Self {
            factors: HashMap::new(),
            ic_calculator: ICRollingCalculator::new(60),
        }
    }

    /// 注册因子
    pub fn register<F: Factor + 'static>(&mut self, factor: F) {
        let name = factor.name().to_string();
        self.factors.insert(name, Box::new(factor));
    }

    /// 获取因子
    pub fn get(&self, name: &str) -> Option<&dyn Factor> {
        self.factors.get(name).map(|b| b.as_ref())
    }

    /// 批量计算所有因子
    pub fn compute_all(
        &self,
        code: &str,
        date: chrono::NaiveDate,
        kline: &KlineData,
    ) -> HashMap<String, FactorValue> {
        self.factors
            .iter()
            .map(|(name, factor)| {
                let value = factor.compute(code, date, kline);
                (name.clone(), value)
            })
            .collect()
    }

    /// 计算因子 IC
    pub fn calculate_ic(
        &mut self,
        factor_name: &str,
        date: chrono::NaiveDate,
        values: &[FactorValue],
        returns: &[f64],
    ) -> f64 {
        self.ic_calculator.calculate_ic(factor_name, date, values, returns)
    }

    /// 获取因子有效性
    pub fn is_decay(&self, factor_name: &str) -> bool {
        self.ic_calculator.is_factor_decay(factor_name)
    }
}

impl Default for FactorLibrary {
    fn default() -> Self {
        Self::new()
    }
}
```

### 3.2 成本模型

```rust
// src/backtest/cost.rs

use serde::{Deserialize, Serialize};

/// 成本配置
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CostConfig {
    /// 印花税 (卖出时收取)
    pub stamp_tax: f64,        // 默认 0.001 (0.1%)
    /// 佣金 (双向收取)
    pub commission: f64,       // 默认 0.0003 (0.03%)
    /// 最低佣金 (每笔)
    pub min_commission: f64,   // 默认 5.0
    /// 滑点 (basis points)
    pub slippage_bps: f64,     // 默认 1.0 bp
    /// 是否模拟冲击成本
    pub market_impact: bool,
}

impl Default for CostConfig {
    fn default() -> Self {
        Self {
            stamp_tax: 0.001,
            commission: 0.0003,
            min_commission: 5.0,
            slippage_bps: 1.0,
            market_impact: false,
        }
    }
}

/// 成本计算器
pub struct CostCalculator {
    config: CostConfig,
}

impl CostCalculator {
    pub fn new(config: CostConfig) -> Self {
        Self { config }
    }

    /// 计算交易成本
    pub fn calculate(&self, fill_price: f64, quantity: i32, side: OrderSide) -> CostResult {
        let turnover = fill_price * quantity as f64;

        // 佣金 (双向)
        let commission = (turnover * self.config.commission)
            .max(self.config.min_commission);

        // 印花税 (仅卖出)
        let stamp_tax = if side == OrderSide::Sell {
            turnover * self.config.stamp_tax
        } else {
            0.0
        };

        // 滑点
        let slippage = fill_price * quantity as f64 * self.config.slippage_bps / 10000.0;

        // 冲击成本 (简化模型)
        let market_impact = if self.config.market_impact {
            turnover * 0.0001 // 假设 1bp 冲击
        } else {
            0.0
        };

        let total_cost = commission + stamp_tax + slippage + market_impact;

        CostResult {
            commission,
            stamp_tax,
            slippage,
            market_impact,
            total_cost,
            net_value: turnover - total_cost,
        }
    }

    /// 计算成交价格 (含滑点)
    pub fn apply_slippage(&self, base_price: f64, side: OrderSide) -> f64 {
        let sign = if side == OrderSide::Buy { 1.0 } else { -1.0 };
        let slippage = base_price * self.config.slippage_bps / 10000.0;
        base_price + sign * slippage
    }
}

/// 成本计算结果
#[derive(Debug, Clone)]
pub struct CostResult {
    pub commission: f64,
    pub stamp_tax: f64,
    pub slippage: f64,
    pub market_impact: f64,
    pub total_cost: f64,
    pub net_value: f64, // 净到手金额
}
```

```rust
// src/backtest/limit_up.rs

use serde::{Deserialize, Serialize};

/// 涨跌停行为配置
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum LimitUpBehavior {
    /// 跳过信号
    Skip,
    /// 部分成交
    PartialFill { fill_rate: f64 },
    /// 次日开盘成交
    NextOpen,
}

impl Default for LimitUpBehavior {
    fn default() -> Self {
        Self::Skip
    }
}

/// 涨跌停检查器
pub struct LimitUpChecker {
    behavior: LimitUpBehavior,
}

impl LimitUpChecker {
    pub fn new(behavior: LimitUpBehavior) -> Self {
        Self { behavior }
    }

    /// 检查订单是否可执行
    pub fn check(&self, code: &str, bar: &Bar, side: OrderSide) -> LimitUpResult {
        let limit_up_price = bar.close * 1.10; // 假设10%涨跌停
        let limit_down_price = bar.close * 0.90;

        match side {
            OrderSide::Buy => {
                if bar.is_suspended {
                    return LimitUpResult::Cannot {
                        reason: "停牌".to_string(),
                        can_retry: false,
                    };
                }
                if bar.is_limit_up {
                    match &self.behavior {
                        LimitUpBehavior::Skip => {
                            return LimitUpResult::Cannot {
                                reason: "涨停无法买入".to_string(),
                                can_retry: true,
                            };
                        }
                        LimitUpBehavior::PartialFill { fill_rate } => {
                            return LimitUpResult::Partial {
                                fill_rate: *fill_rate,
                                reason: format!("涨停部分成交 {:.0}%", fill_rate * 100.0),
                            };
                        }
                        LimitUpBehavior::NextOpen => {
                            return LimitUpResult::Retry {
                                reason: "涨跌停等次日".to_string(),
                            };
                        }
                    }
                }
            }
            OrderSide::Sell => {
                if bar.is_suspended {
                    return LimitUpResult::Cannot {
                        reason: "停牌".to_string(),
                        can_retry: false,
                    };
                }
                if bar.is_limit_down {
                    return LimitUpResult::Cannot {
                        reason: "跌停无法卖出".to_string(),
                        can_retry: true,
                    };
                }
            }
        }

        LimitUpResult::Can
    }
}

/// 涨跌停检查结果
#[derive(Debug, Clone)]
pub enum LimitUpResult {
    /// 可以正常交易
    Can,
    /// 无法交易
    Cannot { reason: String, can_retry: bool },
    /// 部分成交
    Partial { fill_rate: f64, reason: String },
    /// 需要重试 (等下一个bar)
    Retry { reason: String },
}
```

### 3.4 回测引擎核心

```rust
// src/backtest/engine.rs

use super::{
    Bar, CostCalculator, CostConfig, FactorLibrary, Fill, LimitUpBehavior,
    LimitUpChecker, LimitUpResult, MarketEvent, Order, OrderSide, OrderType, PositionUpdate,
    Signal, SignalType,
};
use crate::data_provider::KlineData;
use chrono::NaiveDate;
use log::{info, warn};
use std::collections::HashMap;

/// 回测引擎配置
#[derive(Debug, Clone)]
pub struct EngineConfig {
    /// 初始资金
    pub initial_capital: f64,
    /// 成本配置
    pub cost_config: CostConfig,
    /// 涨跌停行为
    pub limit_up_behavior: LimitUpBehavior,
    /// 最大持仓数
    pub max_positions: usize,
    /// 单票最大仓位比例
    pub max_position_ratio: f64,
    /// 是否允许做空
    pub allow_short: bool,
}

impl Default for EngineConfig {
    fn default() -> Self {
        Self {
            initial_capital: 1_000_000.0, // 100万
            cost_config: CostConfig::default(),
            limit_up_behavior: LimitUpBehavior::Skip,
            max_positions: 10,
            max_position_ratio: 0.2, // 单票最多20%
            allow_short: false,
        }
    }
}

/// 回测引擎
pub struct BacktestEngine {
    config: EngineConfig,
    cash: f64,
    positions: HashMap<String, Position>, // code -> Position
    cost_calculator: CostCalculator,
    limit_up_checker: LimitUpChecker,
    trades: Vec<Fill>,
    signals: Vec<Signal>,
    daily_nav: Vec<(NaiveDate, f64)>,
    events: Vec<MarketEvent>,
}

/// 持仓
#[derive(Debug, Clone)]
struct Position {
    code: String,
    shares: f64,
    avg_cost: f64,
}

impl BacktestEngine {
    pub fn new(config: EngineConfig) -> Self {
        Self {
            cost_calculator: CostCalculator::new(config.cost_config.clone()),
            limit_up_checker: LimitUpChecker::new(config.limit_up_behavior.clone()),
            cash: config.initial_capital,
            config,
            positions: HashMap::new(),
            trades: Vec::new(),
            signals: Vec::new(),
            daily_nav: Vec::new(),
            events: Vec::new(),
        }
    }

    /// 处理单只股票的K线
    pub fn process_bar(&mut self, bar: &Bar, signal: Option<Signal>) {
        // 1. 检查涨跌停
        if let Some(sig) = &signal {
            let limit_result = self.limit_up_checker.check(
                &sig.code,
                bar,
                match sig.signal_type {
                    SignalType::Buy => OrderSide::Buy,
                    _ => OrderSide::Sell,
                },
            );

            match limit_result {
                LimitUpResult::Cannot { reason, .. } => {
                    warn!("[{}] 信号跳过: {}", bar.code, reason);
                    return;
                }
                LimitUpResult::Partial { fill_rate, .. } => {
                    warn!("[{}] 部分成交: {:.0}%", bar.code, fill_rate * 100.0);
                    // 继续执行，按比例成交
                }
                LimitUpResult::Retry { reason } => {
                    warn!("[{}] 信号等待: {}", bar.code, reason);
                    return;
                }
                LimitUpResult::Can => {}
            }
        }

        // 2. 处理信号
        if let Some(sig) = signal {
            self.signals.push(sig.clone());
            self.process_signal(&sig, bar);
        }

        // 3. 记录净值
        self.record_nav(bar.date);
    }

    /// 处理信号
    fn process_signal(&mut self, signal: &Signal, bar: &Bar) {
        match signal.signal_type {
            SignalType::Buy => {
                self.execute_buy(signal, bar);
            }
            SignalType::Sell => {
                self.execute_sell(signal, bar);
            }
            SignalType::SellAll => {
                if let Some(pos) = self.positions.get(&signal.code) {
                    let sell_sig = Signal::sell(
                        &signal.code,
                        bar.close,
                        pos.shares as i32,
                        &signal.reason,
                    );
                    self.execute_sell(&sell_sig, bar);
                }
            }
        }
    }

    /// 买入
    fn execute_buy(&mut self, signal: &Signal, bar: &Bar) {
        // 计算可买数量
        let max_shares = (self.cash * self.config.max_position_ratio / bar.close) as i32;
        let shares = (max_shares / 100 * 100).min(signal.quantity); // 整手

        if shares <= 0 {
            warn!("[{}] 资金不足买入", bar.code);
            return;
        }

        // 成交价格 (含滑点)
        let fill_price = self.cost_calculator.apply_slippage(bar.close, OrderSide::Buy);
        let cost = self.cost_calculator.calculate(fill_price, shares, OrderSide::Buy);

        // 扣除成本
        let total_cost = fill_price * shares as f64 + cost.total_cost;
        if total_cost > self.cash {
            warn!("[{}] 资金不足: 需要 {:.2}, 剩余 {:.2}", bar.code, total_cost, self.cash);
            return;
        }
        self.cash -= total_cost;

        // 更新持仓
        let position = self.positions.entry(bar.code.clone()).or_insert_with(|| {
            Position { code: bar.code.clone(), shares: 0.0, avg_cost: 0.0 }
        });

        let total_cost_basis = position.avg_cost * position.shares + fill_price * shares as f64;
        position.shares += shares as f64;
        position.avg_cost = total_cost_basis / position.shares;

        // 记录成交
        let fill = Fill {
            order_id: signal.id.clone(),
            code: bar.code.clone(),
            side: OrderSide::Buy,
            price: fill_price,
            quantity: shares,
            slippage: cost.slippage,
            commission: cost.commission,
            timestamp: signal.timestamp,
        };
        self.trades.push(fill);

        info!(
            "[{}] 买入 {} 股 @ {:.2}, 成本 {:.2}",
            bar.code, shares, fill_price, total_cost
        );
    }

    /// 卖出
    fn execute_sell(&mut self, signal: &Signal, bar: &Bar) {
        let position = match self.positions.get_mut(&signal.code) {
            Some(p) => p,
            None => {
                warn!("[{}] 无持仓可卖", bar.code);
                return;
            }
        };

        let shares = signal.quantity.min(position.shares as i32);
        if shares <= 0 {
            return;
        }

        // 成交价格 (含滑点)
        let fill_price = self.cost_calculator.apply_slippage(bar.close, OrderSide::Sell);
        let cost = self.cost_calculator.calculate(fill_price, shares, OrderSide::Sell);

        // 收入
        let proceeds = fill_price * shares as f64 - cost.total_cost;
        self.cash += proceeds;

        // 更新持仓
        position.shares -= shares as f64;
        if position.shares < 1.0 {
            self.positions.remove(&signal.code);
        }

        // 记录成交
        let fill = Fill {
            order_id: signal.id.clone(),
            code: bar.code.clone(),
            side: OrderSide::Sell,
            price: fill_price,
            quantity: shares,
            slippage: cost.slippage,
            commission: cost.commission,
            timestamp: signal.timestamp,
        };
        self.trades.push(fill);

        let pnl = (fill_price - position.avg_cost) * shares as f64;
        info!(
            "[{}] 卖出 {} 股 @ {:.2}, 盈亏 {:+.2}",
            bar.code, shares, fill_price, pnl
        );
    }

    /// 记录每日净值
    fn record_nav(&mut self, date: NaiveDate) {
        let position_value: f64 = self.positions.values().map(|p| p.shares * p.avg_cost).sum();
        let total_value = self.cash + position_value;
        self.daily_nav.push((date, total_value));
    }

    /// 获取当前持仓
    pub fn get_positions(&self) -> Vec<PositionUpdate> {
        self.positions
            .values()
            .map(|p| PositionUpdate {
                code: p.code.clone(),
                shares: p.shares,
                avg_cost: p.avg_cost,
                market_value: p.shares * p.avg_cost, // 简化: 用成本价
                unrealized_pnl: 0.0, // 简化
            })
            .collect()
    }

    /// 获取交易记录
    pub fn get_trades(&self) -> &[Fill] {
        &self.trades
    }

    /// 获取净值曲线
    pub fn get_nav(&self) -> &[(NaiveDate, f64)] {
        &self.daily_nav
    }

    /// 获取账户快照
    pub fn get_account(&self) -> AccountUpdate {
        let positions = self.get_positions();
        let position_value: f64 = positions.iter().map(|p| p.market_value).sum();
        AccountUpdate {
            cash: self.cash,
            total_value: self.cash + position_value,
            total_pnl: self.cash + position_value - self.config.initial_capital,
            positions,
        }
    }

    /// 计算业绩指标
    pub fn calc_performance(&self) -> BacktestPerformance {
        let initial = self.config.initial_capital;
        let final_value = self.daily_nav.last().map(|(_, v)| *v).unwrap_or(initial);
        let total_return = (final_value - initial) / initial;

        // 简化夏普: (日收益均值 / 日收益标准差) * sqrt(252)
        let returns: Vec<f64> = self.daily_nav
            .windows(2)
            .map(|w| (w[1].1 - w[0].1) / w[0].1)
            .collect();

        let mean_return = if !returns.is_empty() {
            returns.iter().sum::<f64>() / returns.len() as f64
        } else {
            0.0
        };

        let std_return = if returns.len() > 1 {
            let variance = returns.iter().map(|r| (r - mean_return).powi(2)).sum::<f64>() / returns.len() as f64;
            variance.sqrt()
        } else {
            0.0
        };

        let sharpe = if std_return > 0.0 {
            mean_return / std_return * (252.0 as f64).sqrt()
        } else {
            0.0
        };

        // 最大回撤
        let mut max_value = initial;
        let mut max_drawdown = 0.0;
        for &(_, value) in &self.daily_nav {
            max_value = max_value.max(value);
            let drawdown = (max_value - value) / max_value;
            max_drawdown = max_drawdown.max(drawdown);
        }

        BacktestPerformance {
            total_return,
            annual_return: total_return, // 简化
            sharpe_ratio: sharpe,
            max_drawdown,
            total_trades: self.trades.len(),
            final_value,
        }
    }
}

/// 业绩指标
#[derive(Debug)]
pub struct BacktestPerformance {
    pub total_return: f64,
    pub annual_return: f64,
    pub sharpe_ratio: f64,
    pub max_drawdown: f64,
    pub total_trades: usize,
    pub final_value: f64,
}
```
