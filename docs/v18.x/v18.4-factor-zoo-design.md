## 二、因子计算引擎（polars 批量计算）

```rust
use polars::prelude::*;

/// 因子计算核心引擎
/// 
/// 用 polars 一次性批量计算所有因子
/// 对比逐个 factor.compute() 快 10-100 倍
pub struct FactorEngine<'a> {
    df: &'a DataFrame,  // 已加载的股票历史数据
}

impl<'a> FactorEngine<'a> {
    pub fn new(df: &'a DataFrame) -> Self {
        Self { df }
    }

    /// 一次性计算所有量价因子
    pub fn compute_price_factors(&self) -> Result<DataFrame> {
        self.df.clone()
            .lazy()
            .sort(["code", "date"], Default::default())
            .with_columns([
                // ===== 动量 =====
                col("close")
                    .pct_change(lit(5))
                    .over([col("code")])
                    .alias("momentum_5d"),
                col("close")
                    .pct_change(lit(20))
                    .over([col("code")])
                    .alias("momentum_20d"),
                col("close")
                    .pct_change(lit(60))
                    .over([col("code")])
                    .alias("momentum_60d"),

                // ===== 反转 =====
                (-col("close").pct_change(lit(5)).over([col("code")]))
                    .alias("reversal_5d"),

                // ===== 波动率 =====
                col("close")
                    .pct_change(lit(1))
                    .over([col("code")])
                    .rolling_std(RollingOptionsFixedWindow {
                        window_size: 20,
                        min_periods: 10,
                        ..Default::default()
                    })
                    .alias("volatility_20d"),

                // ===== 下行波动率 =====
                when(col("close").pct_change(1).lt(0))
                    .then(col("close").pct_change(1))
                    .otherwise(lit(0))
                    .rolling_std(RollingOptionsFixedWindow {
                        window_size: 20,
                        min_periods: 10,
                        ..Default::default()
                    })
                    .alias("downside_vol"),

                // ===== 均线偏离 =====
                ((col("close") 
                    - col("close").rolling_mean(
                        RollingOptionsFixedWindow { window_size: 20, min_periods: 5, ..Default::default() }
                    ).over([col("code")]))
                    / col("close").rolling_mean(
                        RollingOptionsFixedWindow { window_size: 20, min_periods: 5, ..Default::default() }
                    ).over([col("code")])
                    * lit(100.0))
                    .alias("ma20_deviation"),

                // ===== 均线交叉 =====
                when(
                    col("close").rolling_mean(
                        RollingOptionsFixedWindow { window_size: 5, min_periods: 1, ..Default::default() }
                    ).over([col("code")])
                    .gt(
                        col("close").rolling_mean(
                            RollingOptionsFixedWindow { window_size: 20, min_periods: 1, ..Default::default() }
                        ).over([col("code")])
                    )
                )
                .then(lit(1.0))
                .otherwise(lit(0.0))
                .alias("ma_cross_5x20"),

                // ===== 量比 =====
                (col("volume") / col("volume").rolling_mean(
                    RollingOptionsFixedWindow { window_size: 20, min_periods: 5, ..Default::default() }
                ).over([col("code")]))
                .alias("volume_ratio_20d"),

                // ===== 放量突破 =====
                when(col("volume").gt(
                    col("volume").rolling_mean(
                        RollingOptionsFixedWindow { window_size: 20, min_periods: 5, ..Default::default() }
                    ).over([col("code")]) * lit(3.0)
                ))
                .then(lit(1.0))
                .otherwise(lit(0.0))
                .alias("volume_breakout"),

                // ===== RSI(14) =====
                Self::rsi_expr(14).alias("rsi_14"),

                // ===== 布林带位置 =====
                Self::bollinger_position_expr(20).alias("bollinger_position"),

                // ===== 新高 =====
                when(col("close").eq(
                    col("close").rolling_max(
                        RollingOptionsFixedWindow { window_size: 20, min_periods: 1, ..Default::default() }
                    ).over([col("code")])
                ))
                .then(lit(1.0))
                .otherwise(lit(0.0))
                .alias("new_high_20"),

                // ===== 换手加速 =====
                when(col("turnover_rate").gt(
                    col("turnover_rate").rolling_mean(
                        RollingOptionsFixedWindow { window_size: 5, min_periods: 1, ..Default::default() }
                    ).over([col("code")]) * lit(2.0)
                ))
                .then(lit(1.0))
                .otherwise(lit(0.0))
                .alias("turnover_acceleration"),

                // ===== 连续涨跌 =====
                col("close")
                    .pct_change(lit(1))
                    .over([col("code")])
                    .gt(lit(0))
                    .cast(DataType::Int32)
                    .alias("up_day"),
            ])
            .collect()
    }

    /// RSI 用 polars 表达式实现 (比 ta crate 快)
    fn rsi_expr(period: usize) -> Expr {
        let change = col("close").diff(1, Default::default()).over([col("code")]);
        let gain = when(change.clone().gt(0))
            .then(change.clone())
            .otherwise(lit(0.0));
        let loss = when(change.clone().lt(0))
            .then(change.clone().abs())
            .otherwise(lit(0.0));

        let avg_gain = gain.rolling_mean(RollingOptionsFixedWindow {
            window_size: period,
            min_periods: period / 2,
            ..Default::default()
        });
        let avg_loss = loss.rolling_mean(RollingOptionsFixedWindow {
            window_size: period,
            min_periods: period / 2,
            ..Default::default()
        });

        when(avg_loss.clone().gt(0))
            .then(lit(100.0) - (lit(100.0) / (lit(1.0) + avg_gain / avg_loss)))
            .otherwise(lit(50.0))
    }

    /// 布林带位置
    fn bollinger_position_expr(period: usize) -> Expr {
        let ma = col("close").rolling_mean(RollingOptionsFixedWindow {
            window_size: period, min_periods: period / 2, ..Default::default()
        }).over([col("code")]);
        let std = col("close").rolling_std(RollingOptionsFixedWindow {
            window_size: period, min_periods: period / 2, ..Default::default()
        }).over([col("code")]);

        let upper = ma.clone() + lit(2.0) * std.clone();
        let lower = ma - lit(2.0) * std.clone();

        when(upper.clone().neq(lower.clone()))
            .then((col("close") - lower.clone()) / (upper - lower))
            .otherwise(lit(0.5))
    }
}

/// 因子质量检查
pub struct FactorQualityCheck {
    pub min_coverage: f64,        // 最小覆盖率 (如 0.8 = 80% 股票有值)
    pub max_outlier_ratio: f64,   // 最大离群值比例
    pub winsorize_sigma: f64,     // 截尾阈值 (如 3.0 = 3σ)
}

impl FactorQualityCheck {
    /// 因子值清洗：去极值 + 标准化 + 缺失值处理
    pub fn clean(&self, values: &mut [f64]) {
        let n = values.len();

        // 1. 统计
        let valid: Vec<f64> = values.iter()
            .filter(|v| v.is_finite())
            .copied()
            .collect();

        if valid.len() < (n as f64 * self.min_coverage) as usize {
            // 覆盖率不足，该因子不可用
            for v in values.iter_mut() {
                *v = 0.0;
            }
            return;
        }

        // 2. Winsorize (MAD方法，比 Z-Score 更鲁棒)
        let median = Self::median(&valid);
        let mad = Self::mad(&valid, median);  // Median Absolute Deviation

        if mad > 1e-10 {
            let upper = median + self.winsorize_sigma * mad;
            let lower = median - self.winsorize_sigma * mad;
            for v in values.iter_mut() {
                if v.is_finite() {
                    *v = v.clamp(lower, upper);
                }
            }
        }

        // 3. Z-Score 标准化
        let mean = values.iter().filter(|v| v.is_finite()).sum::<f64>()
            / valid.len() as f64;
        let std = (values.iter()
            .filter(|v| v.is_finite())
            .map(|v| (v - mean).powi(2))
            .sum::<f64>() / valid.len() as f64)
            .sqrt();

        if std > 1e-10 {
            for v in values.iter_mut() {
                if v.is_finite() {
                    *v = (*v - mean) / std;
                } else {
                    *v = 0.0; // 缺失填0 (中性)
                }
            }
        }

        // 4. 处理方向：确保高值 = 好
        // (由 factor.direction() 控制，在计算时已经处理)
    }
}
```

## 三、5 种回测模式

### Mode 1: 单因子 IC/IR 分析

```rust
/// 因子有效性检验
///
/// 核心指标：
/// - Rank IC: 横截面排序与未来收益的相关性
/// - IR: IC均值 / IC标准差 (> 0.3 有效)
/// - IC胜率: IC > 0 的天数比例 (> 55% 有效)
/// - 分层收益: Top vs Bottom quantile 差异
///
/// 标准:
///   |IC| > 0.03  → 有一定预测力
///   IR  > 0.3   → 稳定性好
///   IC胜率 > 55% → 方向一致

#[derive(Debug, Serialize)]
pub struct FactorTestResult {
    pub factor_name: String,
    pub period: (NaiveDate, NaiveDate),
    pub stock_count: usize,
    pub trading_days: usize,

    // IC 统计
    pub rank_ic_mean: f64,
    pub rank_ic_std: f64,
    pub ir: f64,                   // IR = IC_mean / IC_std
    pub ic_win_rate: f64,          // IC > 0 的比例
    pub ic_t_stat: f64,            // t = IC_mean / (IC_std / sqrt(n))
    pub ic_significant: bool,      // t > 2.0

    // 分层收益 (5分位)
    pub quantile_returns: Vec<QuantileStats>,
    pub top_minus_bottom: f64,     // Q5 - Q1 年化收益
    pub monotonic: bool,           // Q1<Q2<Q3<Q4<Q5 是否严格单调

    // 稳定性
    pub half_life: f64,            // IC半衰期 (天)
    pub decay_flag: bool,          // 是否衰减 (半衰期 < 30天)

    // 换手率
    pub turnover_1d: f64,          // top quintile 1日换手率
    pub turnover_20d: f64,         // 20日换手率
}

#[derive(Debug, Serialize)]
pub struct QuantileStats {
    pub group: usize,              // Q1-Q5
    pub annual_return: f64,       // 年化收益
    pub sharpe: f64,
    pub max_drawdown: f64,
    pub win_rate: f64,
}
```

### Mode 2: 因子分层回测

```rust
/// 分层回测：按因子值分5组，比较各组收益
///
/// 核心逻辑：
/// 1. 每日按因子值排序
/// 2. 分5组 (Q1最差 → Q5最好)
/// 3. 每组等权持仓，持有N天后调仓
/// 4. 画出5条净值曲线
///
/// 判断标准:
///   Q5净值 >> Q1净值  → 因子有效
///   5条线纠缠         → 因子废了

pub struct QuantileBacktest {
    config: QuantileConfig,
}

#[derive(Debug, Clone)]
pub struct QuantileConfig {
    pub n_quantiles: usize,          // 5
    pub holding_period: usize,       // 持仓天数 (如 20)
    pub rebalance_freq: RebalanceFreq,
    pub commission: f64,             // 0.0015 (双边)
    pub universe: Vec<String>,       // 股票池
}

pub enum RebalanceFreq {
    Daily,
    Weekly,    // 每周一调仓
    Monthly,   // 每月初调仓
}

impl QuantileBacktest {
    pub fn run(
        &self,
        factor_values: &[Vec<f64>],    // [N股票 × T天]
        prices: &[Vec<f64>],            // [N股票 × T天] 复权收盘价
        dates: &[NaiveDate],
    ) -> QuantileResult {

        let n_quantiles = self.config.n_quantiles;
        let n_stocks = factor_values.len();
        let mut nav = vec![vec![1.0]; n_quantiles];  // 每组净值
        let mut current_groups: Vec<Vec<usize>> = vec![vec![]; n_quantiles];

        for t in (self.config.holding_period..dates.len())
            .step_by(self.config.holding_period)
        {
            // 1. 按当日因子值排序
            let mut ranked: Vec<(usize, f64)> = (0..n_stocks)
                .filter_map(|i| {
                    let v = factor_values[i][t - self.config.holding_period];
                    if v.is_finite() { Some((i, v)) } else { None }
                })
                .collect();
            ranked.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(Ordering::Equal));

            // 2. 分组
            let per_group = ranked.len() / n_quantiles;
            for g in 0..n_quantiles {
                let start = g * per_group;
                let end = if g == n_quantiles - 1 { ranked.len() } else { (g + 1) * per_group };
                current_groups[g] = ranked[start..end].iter().map(|(i, _)| *i).collect();
            }

            // 3. 计算每组持有期收益
            for g in 0..n_quantiles {
                let stocks = &current_groups[g];
                if stocks.is_empty() { continue; }

                let avg_return: f64 = stocks.iter()
                    .map(|&i| {
                        let entry = prices[i][t - self.config.holding_period];
                        let exit = prices[i][t];
                        (exit - entry) / entry
                    })
                    .sum::<f64>() / stocks.len() as f64;

                let net_return = avg_return - self.config.commission;
                let last = *nav[g].last().unwrap_or(&1.0);
                nav[g].push(last * (1.0 + net_return));
            }
        }

        // 4. 统计
        let mut quantile_stats = Vec::new();
        for g in 0..n_quantiles {
            let returns: Vec<f64> = nav[g].windows(2)
                .map(|w| (w[1] - w[0]) / w[0])
                .collect();
            let annual_ret = returns.iter().sum::<f64>() / returns.len() as f64 * 252.0;
            let sharpe = if returns.len() > 1 {
                let mean = returns.iter().sum::<f64>() / returns.len() as f64;
                let std = (returns.iter().map(|r| (r - mean).powi(2)).sum::<f64>() / returns.len() as f64).sqrt();
                if std > 0.0 { mean / std * (252.0_f64).sqrt() } else { 0.0 }
            } else { 0.0 };
            quantile_stats.push(QuantileStats {
                group: g + 1,
                annual_return: annual_ret,
                sharpe,
                max_drawdown: calc_max_drawdown(&nav[g]),
                win_rate: returns.iter().filter(|&&r| r > 0.0).count() as f64 / returns.len() as f64,
            });
        }

        let top = quantile_stats.last().unwrap().annual_return;
        let bottom = quantile_stats.first().unwrap().annual_return;

        QuantileResult {
            nav_curves: nav,
            quantile_stats,
            top_minus_bottom: top - bottom,
            monotonic: (1..n_quantiles).all(|g| {
                quantile_stats[g].annual_return > quantile_stats[g - 1].annual_return
            }),
        }
    }
}
```

### Mode 3: 多因子排序选股

```rust
/// 多因子横截面排序选股
///
/// 流程:
/// 1. 因子标准化 (Z-Score + Winsorize)
/// 2. 加权求和 → 综合得分
/// 3. 排序 → 选 Top K
/// 4. 等权持仓，每月调仓
/// 5. Walk-Forward 验证

pub struct MultiFactorSelector {
    factors: Vec<FactorWeight>,    // 因子 + 权重
    top_k: usize,                  // 选多少只
    rebalance: RebalanceFreq,
    quality: FactorQualityCheck,
}

#[derive(Debug, Clone)]
pub struct FactorWeight {
    pub name: String,
    pub weight: f64,         // 权重 (正=正向, 负=反向)
    pub winsorize: f64,      // 截尾阈值 (如 3.0)
}

impl MultiFactorSelector {
    /// 多因子打分
    pub fn score(
        &self,
        factor_data: &HashMap<String, Vec<f64>>,  // 因子名 → [N只股票的当日值]
    ) -> Vec<f64> {
        let n_stocks = factor_data.values().next()
            .map(|v| v.len())
            .unwrap_or(0);
        let mut scores = vec![0.0; n_stocks];

        for fw in &self.factors {
            if let Some(raw_values) = factor_data.get(&fw.name) {
                let mut values = raw_values.clone();

                // Winsorize + Z-Score
                self.quality.clean(&mut values);

                for i in 0..n_stocks {
                    scores[i] += values[i] * fw.weight;
                }
            }
        }
        scores
    }

    /// Walk-Forward 优化 + 验证
    pub fn walk_forward(
        &self,
        factor_data: &HashMap<String, Vec<Vec<f64>>>,  // [因子名 → N股票 × T天]
        prices: &[Vec<f64>],
        dates: &[NaiveDate],
        train_months: usize,   // 训练窗口 (如 12个月)
        test_months: usize,    // 测试窗口 (如 3个月)
    ) -> WalkForwardResult {
        let mut is_results = Vec::new();
        let mut os_results = Vec::new();
        let days_per_month = 21;

        let mut start = 0;
        while start + (train_months + test_months) * days_per_month < dates.len() {
            let train_end = start + train_months * days_per_month;
            let test_end = train_end + test_months * days_per_month;

            // 训练期: 网格搜索最优权重
            let best_weights = self.grid_search_weights(
                factor_data, prices, &dates[start..train_end]
            );

            // 样本内
            let is_perf = self.evaluate_period(
                &best_weights, factor_data, prices, &dates[start..train_end]
            );
            is_results.push(is_perf);

            // 样本外
            let os_perf = self.evaluate_period(
                &best_weights, factor_data, prices, &dates[train_end..test_end]
            );
            os_results.push(os_perf);

            start += test_months * days_per_month;
        }

        let is_sharpe_mean = is_results.iter().map(|r| r.sharpe).sum::<f64>() / is_results.len() as f64;
        let os_sharpe_mean = os_results.iter().map(|r| r.sharpe).sum::<f64>() / os_results.len() as f64;

        WalkForwardResult {
            in_sample_sharpe: is_sharpe_mean,
            out_sample_sharpe: os_sharpe_mean,
            overfit_ratio: if os_sharpe_mean > 0.0 { is_sharpe_mean / os_sharpe_mean } else { f64::INFINITY },
            stable: os_sharpe_mean > 0.5,
            warning: if is_sharpe_mean / os_sharpe_mean.max(0.01) > 2.0 {
                Some("⚠️ 过拟合风险: IS/OS > 2".to_string())
            } else {
                None
            },
        }
    }
}

#[derive(Debug, Serialize)]
pub struct WalkForwardResult {
    pub in_sample_sharpe: f64,
    pub out_sample_sharpe: f64,
    pub overfit_ratio: f64,      // > 2.0 → 过拟合
    pub stable: bool,
    pub warning: Option<String>,
}
```

```rust
/// 因子择时：不同市场环境用不同因子权重
///
/// 市场状态识别:
/// ┌──────────┬──────────────────────┬────────────────────┐
/// │ 状态      │ 条件                  │ 推荐权重           │
/// ├──────────┼──────────────────────┼────────────────────┤
/// │ BullTrend │ 指数 > MA20 + MACD金叉 │ 动量30% + 成长20% │
/// │ BearTrend │ 指数 < MA20 + MACD死叉 │ 低估值30% + 质量20%│
/// │ Sideways  │ 布林带带宽 < 历史中位数 │ 反转25% + 短线20% │
/// │ Panic     │ VIX/波动率 > 2σ        │ 超跌30% + 质量20% │
/// └──────────┴──────────────────────┴────────────────────┘

#[derive(Debug, Clone, Copy, PartialEq, Hash, Eq)]
pub enum MarketRegime {
    Bull,
    Bear,
    Sideways,
    Panic,
    Recovery,
}

impl MarketRegime {
    /// 根据指数数据识别状态
    pub fn detect(index_klines: &[KlineData], idx: usize) -> Self {
        if idx < 60 { return Self::Sideways; }

        let close = index_klines[idx].close;
        let ma20: f64 = index_klines[idx-19..=idx].iter().map(|k| k.close).sum::<f64>() / 20.0;
        let volatility: f64 = (0..20).map(|i| {
            let r = (index_klines[idx-i].close / index_klines[idx-i-1].close - 1.0).abs();
            r
        }).sum::<f64>() / 20.0;

        let above_ma = close > ma20;
        let high_vol = volatility > 0.025; // 日波动 > 2.5%

        if above_ma && close / index_klines[idx-20].close > 1.05 {
            Self::Bull
        } else if !above_ma && close / index_klines[idx-20].close < 0.95 && high_vol {
            Self::Panic
        } else if !above_ma {
            Self::Bear
        } else if volatility < 0.01 {
            Self::Sideways
        } else {
            Self::Recovery
        }
    }
}
```

```
                     v1 (之前)        v2 (本次)

计算方式              逐个             polars批量
```

**关键洞察:**

1. **55个因子够了** — 覆盖量价/基本面/资金流/情绪/事件/行业所有维度
2. **polars批量计算是关键** — 比逐个 compute() 快 100 倍
3. **5种回测模式互补** — IC检验→分层→多因子→信号→择时，层层递进
4. **Walk-Forward是底线** — 不过拟合才有实盘价值
5. **因子择时是进阶** — 牛市用动量，熊市用价值，震荡用反转
