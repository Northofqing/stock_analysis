### 2.2 核心模块设计

#### 2.2.1 Gateway Federation

```rust
// 网关联邦：统一路由 + 熔断 + 降级
pub struct GatewayFederation {
    primary: Box<dyn MarketGateway>,
    fallbacks: Vec<Box<dyn MarketGateway>>,
    circuit_breaker: CircuitBreaker,
    metrics: GatewayMetrics,
}

impl GatewayFederation {
    pub async fn fetch(&self, req: &MarketRequest) -> Result<MarketData> {
        // 1. 检查熔断器
        if self.circuit_breaker.is_open() {
            return self.fallback(req).await;
        }
        
        // 2. 尝试主网关
        match self.primary.fetch(req).await {
            Ok(data) => {
                self.circuit_breaker.record_success();
                return Ok(data);
            }
            Err(e) => {
                self.circuit_breaker.record_failure();
                return self.fallback(req).await;
            }
        }
    }
}
```

);
